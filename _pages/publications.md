<script>
  // Highlight the chip matching the current bibsearch hash filter.
  (function () {
    function syncTopicChips() {
      var current = decodeURIComponent(window.location.hash.substring(1)).toLowerCase();
      document.querySelectorAll("#pub-topics-bar .pub-topic").forEach(function (chip) {
        var label = decodeURIComponent(chip.getAttribute("href").substring(1)).toLowerCase();
        chip.classList.toggle("active", label === current);
      });
    }
    window.addEventListener("hashchange", syncTopicChips);
    document.addEventListener("DOMContentLoaded", syncTopicChips);
  })();
</script>
